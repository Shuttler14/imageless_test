<script>
document.addEventListener('DOMContentLoaded', () => {
  const pendingDesign = localStorage.getItem('mn_pending_design');
  
  if (pendingDesign) {
    try {
      const data = JSON.parse(pendingDesign);
      
      if (data.slogan && (Date.now() - data.timestamp) < 3600000) {
        console.log("🎁 Gift Mode Activated");

        if (typeof state !== 'undefined') {
          state.isGiftMode = true;
          state.giftContext = data.context;
          state.selectedQuote = data.slogan;
          state.topic = `Gift for ${data.context.recipient}`;
          
          // Psychological header change
          setTimeout(() => {
            const titleEl = document.querySelector('#step-3 .cinematic-title');
            const subEl = document.querySelector('#step-3 .cinematic-subtitle');
            
            if (titleEl) titleEl.innerText = "HOW SHOULD THE DESIGN LOOK?";
            if (subEl) subEl.innerText = "We have the message. Now let's give it a visual soul.";
          }, 150);

          // Auto-skip to Step 3
          setTimeout(() => {
            if (typeof goToStep === 'function') goToStep(3);
          }, 100);
        }
      }
      
      localStorage.removeItem('mn_pending_design');
      
    } catch (e) {
      console.error("Error loading gift context:", e);
    }
  }
});
</script>

Update 1: checkAIReady()

Gift Mode only needs Color + Placement (not Vibe)

function checkAIReady() {
    if (state.isGiftMode) {
        // Gift Mode: Only need Color & Placement
        if (state.aiColor && state.aiPlacement) {
            aiGenerateBtn.classList.add('ready');
        } else {
            aiGenerateBtn.classList.remove('ready');
        }
    } else {
        // Standard Mode: Need all 3
        if (state.aiColor && state.aiPlacement && state.aiVibe) {
            aiGenerateBtn.classList.add('ready');
        } else {
            aiGenerateBtn.classList.remove('ready');
        }
    }
}

Update 2: aiChoice Event Listener

Hide Vibe question for Gift Mode

aiChoice.addEventListener('click', () => { 
    aiOverlay.classList.add('active'); 
    preferencesDrawer.classList.remove('expanded'); 
    
    // Hide Vibe question for Gift Mode
    const allQuestions = document.querySelectorAll('.ai-question');
    if (state.isGiftMode && allQuestions.length >= 3) {
        allQuestions[2].style.display = 'none';
        console.log("🎁 Hiding Vibe question for Gift Mode");
    } else if (allQuestions.length >= 3) {
        allQuestions[2].style.display = 'block';
    }
});

Update 3: aiGenerateBtn Event Listener

Use gift context for style generation

aiGenerateBtn.addEventListener('click', () => {
  if(!aiGenerateBtn.classList.contains('ready')) return;
  aiOverlay.classList.remove('active');
  
  state.color = state.aiColor;
  if(state.aiPlacement === 'front') { 
    state.side1.placement = 'chest'; 
  } else { 
    state.side1.placement = 'back-full'; 
  }

  // NEW LOGIC: PREPARE STYLE STRING
  if (state.isGiftMode && state.giftContext) {
      // 🎁 GIFT MODE
      const ctx = state.giftContext;
      state.selectedStyle = `Gift Design for ${ctx.recipient} on ${ctx.occasion}. Context: ${ctx.unspoken || 'None'}. Style: Emotionally resonant and visually premium.`;
      state.toneKey = "emotionally real";
      console.log("🎁 Gift Mode Style:", state.selectedStyle);
  } else {
      // 👤 SELF MODE
      switch(state.aiVibe) {
          case 'calm': 
            state.selectedStyle = "Minimalist Type"; 
            state.toneKey = "stoic"; 
            break;
          case 'bold': 
            state.selectedStyle = "Cyberpunk Neon"; 
            state.toneKey = "unapologetic"; 
            break;
          case 'disciplined': 
            state.selectedStyle = "Minimalist Type"; 
            state.toneKey = "hustle"; 
            break;
          case 'free': 
            state.selectedStyle = "Vintage 90s"; 
            state.toneKey = "retro"; 
            break;
          case 'dark': 
            state.selectedStyle = "Cyberpunk Neon"; 
            state.toneKey = "darkhumour"; 
            break;
          case 'light': 
            state.selectedStyle = "Y2K Chrome"; 
            state.toneKey = "healing"; 
            break;
          default: 
            state.selectedStyle = "Minimalist Type"; 
            state.toneKey = "stoic";
      }
      console.log("👤 Self Mode Style:", state.selectedStyle);
  }
  
  autoGenerate(); 
});
